                            text = "📋 Available Configs:\n" + "\n".join([f"• {n}" for n in names])
                            if len(text) > 4000: text = text[:4000] + "\n\n... (truncated)"
                            send_message(text)
                    elif msg == "/zip": do_zip()
                    elif msg.startswith("/get"):
                        parts = msg.split(" ", 1)
                        if len(parts) < 2: send_message("⚠️ Usage: /get SIAM VPN")
                        else: get_single(parts[1].strip())
                    elif msg == "/check": send_message("✅ Bot is running perfectly in the background.")
                    else: send_message("❓ Unknown command. Type /help")
        except Exception as e:
            print(f"[!] Loop Error: {e}", file=sys.stderr)
            time.sleep(3)
        time.sleep(1)

if __name__ == "__main__":
    main()
