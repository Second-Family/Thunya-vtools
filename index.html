#!/usr/bin/env python3
import os, sys, zipfile, subprocess, requests

# ===== ညီလေး အချက်အလက် ထည့်ပါ =====
BOT_TOKEN = "YOUR_BOT_TOKEN_HERE"
CHAT_ID   = "YOUR_CHAT_ID_HERE"
# =====================================

WORK_DIR = "/opt/xd-test"
OUT_DIR  = os.path.join(WORK_DIR, "output")
ZIP_PATH = os.path.join(WORK_DIR, "vpn_configs.zip")
TOOL     = os.path.join(WORK_DIR, "XD-Tools.py")
PYTHON   = os.path.join(WORK_DIR, "venv", "bin", "python3")

def tg_send_message(text):
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendMessage"
    try:
        r = requests.post(url, data={"chat_id": CHAT_ID, "text": text, "parse_mode": "HTML"}, timeout=30)
        if not r.ok:
            print(f"[!] sendMessage: {r.text}", file=sys.stderr)
    except Exception as e:
        print(f"[!] sendMessage: {e}", file=sys.stderr)

def tg_send_document(path, caption=""):
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/sendDocument"
    try:
        with open(path, "rb") as f:
            r = requests.post(url, data={"chat_id": CHAT_ID, "caption": caption}, files={"document": f}, timeout=300)
        return r.ok
    except Exception as e:
        print(f"[!] sendDocument: {e}", file=sys.stderr)
        return False

def get_names():
    res = subprocess.run([PYTHON, TOOL, "--list"], capture_output=True, text=True)
    if res.returncode != 0:
        tg_send_message(f"❌ List failed:\n<pre>{res.stderr[:500]}</pre>")
        sys.exit(1)
    names = []
    for line in res.stdout.splitlines():
        line = line.strip()
        if not line or line.startswith("["):
            continue
        names.append(line)
    return names

def download_all(names):
    os.makedirs(OUT_DIR, exist_ok=True)
    ok, fail = 0, 0
    for i, name in enumerate(names, 1):
        print(f"[{i}/{len(names)}] {name}")
        r = subprocess.run([PYTHON, TOOL, "--name", name, "--out", OUT_DIR],
                           capture_output=True, text=True)
        if r.returncode == 0:
            ok += 1
        else:
            fail += 1
            print(f"  [!] {name}: {r.stderr.strip()[:150]}")
    return ok, fail

def make_zip():
    if os.path.exists(ZIP_PATH):
        os.remove(ZIP_PATH)
    with zipfile.ZipFile(ZIP_PATH, "w", zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(OUT_DIR):
            for fn in files:
                full = os.path.join(root, fn)
                zf.write(full, os.path.relpath(full, OUT_DIR))
    return os.path.getsize(ZIP_PATH) / (1024 * 1024)

def main():
    tg_send_message("🚀 <b>XD-Tools started</b>")
    names = get_names()
    if not names:
        tg_send_message("❌ No configs.")
        return
    tg_send_message(f"📥 Downloading <b>{len(names)}</b> configs...")
    ok, fail = download_all(names)
    size_mb = make_zip()
    caption = f"✅ <b>VPN Configs</b>\n📦 {ok} ok, {fail} failed\n💾 {size_mb:.2f} MB"
    if size_mb > 50:
        tg_send_message(f"⚠️ ZIP {size_mb:.2f} MB — splitting...")
        part_size = 45 * 1024 * 1024
        with open(ZIP_PATH, "rb") as f:
            idx = 1
            while True:
                chunk = f.read(part_size)
                if not chunk:
                    break
                pp = f"{ZIP_PATH}.part{idx:02d}"
                with open(pp, "wb") as pf:
                    pf.write(chunk)
                tg_send_document(pp, f"Part {idx}\n{caption}")
                os.remove(pp)
                idx += 1
        tg_send_message("✅ All parts sent.")
    else:
        if tg_send_document(ZIP_PATH, caption):
            tg_send_message("✅ <b>Done!</b>")
        else:
            tg_send_message("❌ Upload failed.")

if __name__ == "__main__":
    main()
