#!/usr/bin/env python3
import os, sys, zipfile, subprocess, requests, time

# ===== ညီလေး အချက်အလက် =====
BOT_TOKEN = "8764708526:AAEn-tq7zy1AUdGn8USzl8Eh6ssc6YlY4eU"
CHAT_ID   = "7576434717"
# ==============================

WORK_DIR = "/opt/xd-test"
OUT_DIR  = os.path.join(WORK_DIR, "output")
ZIP_PATH = os.path.join(WORK_DIR, "vpn_configs.zip")
TOOL     = os.path.join(WORK_DIR, "XD-Tools.py")
PYTHON   = os.path.join(WORK_DIR, "venv", "bin", "python3")
OFFSET   = 0

def tg_api(method, data=None, files=None):
    url = f"https://api.telegram.org/bot{BOT_TOKEN}/{method}"
    try:
        r = requests.post(url, data=data, files=files, timeout=60)
        return r.json()
    except Exception as e:
        print(f"[!] API Error: {e}", file=sys.stderr)
        return {"ok": False}

def send_message(text):
    tg_api("sendMessage", data={"chat_id": CHAT_ID, "text": text, "parse_mode": "HTML"})

def send_document(path, caption=""):
    with open(path, "rb") as f:
        tg_api("sendDocument", data={"chat_id": CHAT_ID, "caption": caption}, files={"document": f})

def get_names():
    res = subprocess.run([PYTHON, TOOL, "--list"], capture_output=True, text=True)
    if res.returncode != 0: return []
    names = []
    for line in res.stdout.splitlines():
        line = line.strip()
        if not line or line.startswith("["): continue
        names.append(line)
    return names

def do_zip():
    send_message("⏳ Downloading all configs to ZIP...\nPlease wait...")
    names = get_names()
    os.makedirs(OUT_DIR, exist_ok=True)
    for f in os.listdir(OUT_DIR): os.remove(os.path.join(OUT_DIR, f))
    ok, fail = 0, 0
    for i, name in enumerate(names, 1):
        r = subprocess.run([PYTHON, TOOL, "--name", name, "--out", OUT_DIR], capture_output=True, text=True)
        if r.returncode == 0: ok += 1
        else: fail += 1
    if os.path.exists(ZIP_PATH): os.remove(ZIP_PATH)
    with zipfile.ZipFile(ZIP_PATH, "w", zipfile.ZIP_DEFLATED) as zf:
        for root, _, files in os.walk(OUT_DIR):
            for fn in files:
                full = os.path.join(root, fn)
                zf.write(full, os.path.relpath(full, OUT_DIR))
    size_mb = os.path.getsize(ZIP_PATH) / (1024 * 1024)
    caption = f"✅ VPN Configs ZIP\n📦 {ok} ok, {fail} failed\n💾 {size_mb:.2f} MB"
    if size_mb > 50:
        send_message("⚠️ ZIP > 50MB, splitting...")
        part_size = 45 * 1024 * 1024
        with open(ZIP_PATH, "rb") as f:
            idx = 1
            while True:
                chunk = f.read(part_size)
                if not chunk: break
                pp = f"{ZIP_PATH}.part{idx:02d}"
                with open(pp, "wb") as pf: pf.write(chunk)
                send_document(pp, f"Part {idx}\n{caption}")
                os.remove(pp)
                idx += 1
    else:
        send_document(ZIP_PATH, caption)
        send_message("✅ Done!")

def get_single(name):
    send_message(f"⏳ Fetching: {name}...")
    os.makedirs(OUT_DIR, exist_ok=True)
    r = subprocess.run([PYTHON, TOOL, "--name", name, "--out", OUT_DIR], capture_output=True, text=True)
    if r.returncode != 0:
        send_message(f"❌ Failed: {r.stderr[:200]}")
        return
    files = os.listdir(OUT_DIR)
    if not files:
        send_message("❌ Config file not created.")
        return
    latest = max([os.path.join(OUT_DIR, f) for f in files], key=os.path.getctime)
    ext = os.path.splitext(latest)[1]
    new_path = os.path.join(WORK_DIR, f"{name}{ext}")
    os.rename(latest, new_path)
    send_document(new_path, f"✅ {name}")
    os.remove(new_path)

def main():
    global OFFSET
    send_message("🚀 XD-Tools Bot is Online!\nType /help to see commands.")
    while True:
        try:
            res = tg_api("getUpdates", data={"offset": OFFSET, "timeout": 30})
            if res.get("ok") and res.get("result"):
                for update in res["result"]:
                    OFFSET = update["update_id"] + 1
                    if "message" not in update or "text" not in update["message"]: continue
                    msg = update["message"]["text"].strip()
                    chat_id = str(update["message"]["chat"]["id"])
                    if chat_id != CHAT_ID: continue
                    if msg in ["/start", "/help"]:
                        help_txt = (
                            "🤖 XD-Tools Commands\n\n"
                            "📋 /list - Config နာမည်အားလုံး ကြည့်ရန်\n"
                            "📦 /zip - အားလုံးကို ZIP ဖိုင်နဲ့ ရယူရန်\n"
                            "📄 /get <name> - Config တစ်ခုချင်း ရယူရန်\n"
                            "🔍 /check - လက်ရှိ run နေတဲ့ အနေအထား\n\n"
                            "Example:\n/get SIAM VPN"
                        )
                        send_message(help_txt)
                    elif msg == "/list":
                        send_message("⏳ Fetching list...")
                        names = get_names()
                        if not names: send_message("❌ No configs found.")
                        else:
                            text = "📋 Available Configs:\n" + "\n".join([f"• {n}" for n in names])
                            if len(text) > 4000: text = text[:4000] + "\n\n... (truncated)"
                            send_message(text)
                    elif msg == "/zip": do_zip()
                    elif msg.startswith("/get"):
                        parts = msg.split(" ", 1)
                        if len(parts) < 2: send_message("⚠️ Usage: /get SIAM VPN")
                        else: get_single(parts[1].strip())
                    elif msg == "/check": send_message("✅ Bot is running perfectly in the background.")
                    else: send_message("❓ Unknown command. Type /help")
        except Exception as e:
            print(f"[!] Loop Error: {e}", file=sys.stderr)
            time.sleep(3)
        time.sleep(1)

if __name__ == "__main__":
    main()

အဆင့် ၇ — Bot ကို နောက်ကွယ်မှ အမြဲ Run နေအောင် လုပ်ခြင်း (Systemd)

Terminal ပိတ်လိုက်ရင်လည်း Bot က ဆက်အလုပ်လုပ်နေအောင်၊ VPS ပြန် Restart ချလည်း Bot က အလိုအလျောက် ပြန်တက်လာအောင် System Service အဖြစ် သတ်မှတ်ပါမယ်။
1Service ဖိုင် ဖန်တီးပါ
Terminal 
nano /etc/systemd/system/xdbot.service
# အောက်က ကုဒ်ကို ထည့်ပါ
xdbot.service 
[Unit]
Description=XD-Tools Telegram Bot
After=network.target

[Service]
Type=simple
User=root
WorkingDirectory=/opt/xd-test
ExecStart=/opt/xd-test/venv/bin/python3 /opt/xd-test/bot.py
Restart=always
RestartSec=5
StandardOutput=syslog
StandardError=syslog
SyslogIdentifier=xdbot

[Install]
WantedBy=multi-user.target
2Service ကို စတင် Run ပါ
# Systemd ကို အသစ်ဖန်တီးထားတဲ့ service ဖိုင်ကို ဖတ်ခိုင်းပါ
Terminal 
systemctl daemon-reload
# Bot ကို စတင်ပါ
Terminal 
systemctl start xdbot
# VPS တက်တိုင်း Bot အလိုအလျောက် တက်လာအောင် သတ်မှတ်ပါ
Terminal 
systemctl enable xdbot
✅ အခုဆိုရင် Terminal ကို ပိတ်လိုက်ရင်လည်း Bot က နောက်ကွယ်မှာ အလုပ်လုပ်နေပါပြီ။

အဆင့် ၈ — အဆင့်မြင့် Commands များ (ဖွင့်/ပိတ်/ပြန်စ)

Terminal ထဲမှာ အောက်ပါ Commands တွေ ရိုက်ပြီး Bot ကို ထိန်းချုပ်လို့ရပါပြီ။

1Bot ရဲ့ အခြေအနေ (Status) ကြည့်ရန်
Terminal 
systemctl status xdbot

(အစိမ်းရောင် Active: running ဆိုရင် Bot အလုပ်လုပ်နေတယ်လို့ ဆိုလိုပါတယ်)

2Bot ကို ရပ်ချင်ရင် (Stop)
Terminal 
systemctl stop xdbot
3Bot ကို ပြန်စချင်ရင် (Restart)
Terminal 
systemctl restart xdbot

Terminal 
journalctl -u xdbot -f
